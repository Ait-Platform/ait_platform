<script>
    let currentSlide = 0;
    const totalSlides = 13;
    let sessionState = 'lobby';
    let pollInterval = null;

    function startWorkshop() {
        fetch('/sace/workshop/start', {method: 'POST'})
            .then(() => fetchState());
    }

    function resetWorkshop() {
        if(confirm("Are you sure? This resets all connected devices back to the lobby.")) {
            fetch('/sace/workshop/reset', {method: 'POST'})
                .then(() => fetchState());
        }
    }

    function fetchState() {
        fetch('/sace/workshop/get_state')
            .then(res => res.json())
            .then(data => {
                sessionState = data.status;
                currentSlide = data.slide;
                document.getElementById('attendance-counter').innerText = data.attendance;
                updateView();
            });
    }

    // Auto-poll state every 2 seconds to get live attendance
    setInterval(fetchState, 2000);

    function updateView() {
        // Pause any currently playing audio
        document.querySelectorAll('audio').forEach(audio => {
            audio.pause();
            audio.currentTime = 0;
        });

        document.querySelectorAll('.slide-container').forEach(el => el.classList.remove('flex'));
        document.querySelectorAll('.slide-container').forEach(el => el.classList.add('hidden'));

        if (sessionState === 'lobby') {
            document.getElementById('controls-lobby').classList.remove('hidden');
            document.getElementById('controls-active').classList.add('hidden');
            const lobbySlide = document.getElementById('slide-lobby');
            lobbySlide.classList.remove('hidden');
            lobbySlide.classList.add('flex');
            document.getElementById('slide-counter').innerText = "Lobby";
            return;
        }

        // ACTIVE STATE
        document.getElementById('controls-lobby').classList.add('hidden');
        document.getElementById('controls-active').classList.remove('hidden');

        const activeSlide = document.getElementById('slide-' + currentSlide);
        if (activeSlide) {
            activeSlide.classList.remove('hidden');
            activeSlide.classList.add('flex');
        }

        // Play audio 1 second after slide reveals
        if (activeSlide) {
            const audioEl = activeSlide.querySelector('audio');
            if (audioEl) {
                setTimeout(() => {
                    audioEl.play().catch(e => console.log('Autoplay blocked by browser:', e));
                }, 1000);
            }
        }

        // Update counter
        const counterEl = document.getElementById('slide-counter');
        if (counterEl) counterEl.innerText = currentSlide + " / " + totalSlides;

        // Sync with server (if we changed slide locally)
        fetch('/sace/workshop/set_slide', {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify({slide: currentSlide})
        });
    }

    function prevSlide() {
        if (currentSlide > 0) {
            currentSlide--;
            updateView();
        }
    }

    function nextSlide() {
        if (currentSlide < totalSlides) {
            currentSlide++;
            updateView();
        }
    }
    
    // Initialize
    fetchState();
</script>
