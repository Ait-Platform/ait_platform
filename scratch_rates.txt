  let masterOwners = [];

  // --- OWNERS (STEP 12) ---
  function buildOwnersDashboard() {
    const masterContainer = document.getElementById('owners-master-container');
    const addressContainer = document.getElementById('addresses-master-container');
    const assignmentContainer = document.getElementById('owners-assignment-container');
    masterContainer.innerHTML = '';
    addressContainer.innerHTML = '';
    assignmentContainer.innerHTML = '';
    
    // Extract unique owners and addresses from wizardData.owners
    masterOwners = [];
    masterAddresses = [];
    if (wizardData.owners && wizardData.owners.length > 0) {
        let uniqueOwnersKeys = new Set();
        let uniqueAddrKeys = new Set();
        wizardData.owners.forEach(o => {
            let ownerKey = (o.name || '') + '|' + (o.email || '');
            if (!uniqueOwnersKeys.has(ownerKey) && (o.name || o.email)) {
                uniqueOwnersKeys.add(ownerKey);
                masterOwners.push({ name: o.name || '', email: o.email || '' });
            }
            
            let addrKey = (o.address || '');
            if (!uniqueAddrKeys.has(addrKey) && o.address) {
                uniqueAddrKeys.add(addrKey);
                masterAddresses.push({ address: o.address || '' });
            }
        });
    }
    
    // Pad to expected counts
    while (masterOwners.length < EXPECTED_OWNERS) {
        masterOwners.push({ name: '', email: '' });
    }
    while (masterAddresses.length < EXPECTED_ADDRESSES) {
        masterAddresses.push({ address: '' });
    }

    renderMasterOwners();
    renderMasterAddresses();
    renderOwnerAssignments();
  }

  window.addOwnerRow = function() {

  window.addAddressRow = function() {
    masterAddresses.push({ address: '' });
    renderMasterAddresses();
    renderOwnerAssignments(); // update dropdowns
  };

  window.updateMasterOwner = function(index, field, value) {
    masterOwners[index][field] = value;
    renderOwnerAssignments();
    triggerAutoSave();
  };

  window.updateMasterAddress = function(index, value) {
    masterAddresses[index].address = value;
    renderOwnerAssignments();
    triggerAutoSave();
  };

  function renderOwnerAssignments() {
    const assignmentContainer = document.getElementById('owners-assignment-container');
    assignmentContainer.innerHTML = '';
    
    wizardData.accounts.forEach(acc => {
      if(!acc.number) return;
      
      const savedOwn = (wizardData.owners && wizardData.owners.find(x => x.account_id === acc.id));
      let selectedOwnerIdx = -1;
      let selectedAddrIdx = -1;
      if (savedOwn) {
          selectedOwnerIdx = masterOwners.findIndex(m => m.name === savedOwn.name && m.email === savedOwn.email);
          if (PROPERTY_ADDRESS.trim() !== '' && savedOwn.address === PROPERTY_ADDRESS) {
              selectedAddrIdx = -2;
          } else {
              selectedAddrIdx = masterAddresses.findIndex(m => m.address === savedOwn.address);
          }
      }
      
      let cardColor = acc.isBulk ? 'bg-amber-50 border-amber-300' : 'bg-sky-50 border-sky-300';
      
      let ownerOptionsHtml = '<option value="-1">-- Select Owner --</option>';
      masterOwners.forEach((m, idx) => {
          let label = m.name || `Owner ${idx + 1}`;
          ownerOptionsHtml += `<option value="${idx}" ${selectedOwnerIdx === idx ? 'selected' : ''}>${label}</option>`;
      });

      let addrOptionsHtml = '<option value="-1">-- Select Address --</option>';
      if (PROPERTY_ADDRESS.trim() !== '') {
          addrOptionsHtml += `<option value="-2" ${selectedAddrIdx === -2 ? 'selected' : ''}>Use Property Physical Address</option>`;
      }
      masterAddresses.forEach((m, idx) => {
          let label = m.address || `Address ${idx + 1}`;
          addrOptionsHtml += `<option value="${idx}" ${selectedAddrIdx === idx ? 'selected' : ''}>${label}</option>`;
      });

      assignmentContainer.innerHTML += `
        <div class="flex items-center justify-between p-3 border rounded-lg shadow-sm own-assign-card ${cardColor}" data-acc-id="${acc.id}">
          <span class="font-bold text-slate-800 w-1/4">${acc.number} ${acc.isBulk ? '<span class="text-[10px] bg-amber-200 text-amber-800 px-2 py-0.5 rounded ml-2 font-bold uppercase border border-amber-300 shadow-sm">Bulk</span>' : '<span class="text-[10px] bg-sky-200 text-sky-800 px-2 py-0.5 rounded ml-2 font-bold uppercase border border-sky-300 shadow-sm">Sub</span>'}</span>
          <div class="w-3/4 flex space-x-2">
            <select class="own-assign-select-owner w-1/2 rounded border border-slate-400 bg-white px-3 py-1.5 text-sm outline-none focus:border-blue-500 focus:ring-2 focus:ring-blue-100 shadow-inner transition" onchange="triggerAutoSave()">
                ${ownerOptionsHtml}
            </select>
            <select class="own-assign-select-addr w-1/2 rounded border border-slate-400 bg-white px-3 py-1.5 text-sm outline-none focus:border-blue-500 focus:ring-2 focus:ring-blue-100 shadow-inner transition" onchange="triggerAutoSave()">
                ${addrOptionsHtml}
            </select>
          </div>
        </div>
      `;
    });
  }

  function gatherOwners() {
    if (currentStep < 12) return true;
    wizardData.owners = [];
    document.querySelectorAll('.own-assign-card').forEach(card => {
      let accId = card.dataset.accId;
      let ownerSelect = card.querySelector('.own-assign-select-owner');
      let addrSelect = card.querySelector('.own-assign-select-addr');
      let oIdx = parseInt(ownerSelect.value);
      let aIdx = parseInt(addrSelect.value);
      
      let oName = '', oEmail = '', oAddr = '';
      if (oIdx >= 0 && masterOwners[oIdx]) {
          oName = masterOwners[oIdx].name;
          oEmail = masterOwners[oIdx].email;
      }
      if (aIdx === -2) {
          oAddr = PROPERTY_ADDRESS;
      } else if (aIdx >= 0 && masterAddresses[aIdx]) {
          oAddr = masterAddresses[aIdx].address;
      }
      
      wizardData.owners.push({
          account_id: accId,
          name: oName,
          email: oEmail,
          address: oAddr
      });
    });
    return true;
  }

  async function saveArchitecture() {
    if(!gatherMapping() || !gatherInitialReadings() || !gatherArrears() || !gatherRates() || !gatherArrangements() || !gatherOwners()) {
      return; // Do not save if validation fails on the final page
    }
    console.log("Saving Architecture:", wizardData);
    
    try {
      const response = await fetch("{{ url_for('billing_bp.save_global_architecture', property_id=property.id) }}", {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-CSRFToken': '{{ csrf_token() }}'
        },
        body: JSON.stringify(wizardData)
      });
      
      if (response.ok) {
        // Clear draft on successful save!
        window.location.href = "{{ url_for('billing_bp.architecture_summary', property_id=property.id) }}?from_wizard=1";
      } else {
        const errData = await response.json();
        alert("Server Error: " + (errData.error || "Please ensure all data is valid."));
      }
    } catch (e) {
      console.error(e);
      alert("Network error: " + e.message);
    }
  }
